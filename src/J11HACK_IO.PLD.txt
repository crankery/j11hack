/* *************** EQUATIONS *********************/

DA7      = !LA2 & DR0 & SLU0 # LA2 & TBRE0 & SLU0 # !LA2 & DR1 & SLU1 # LA2 & TBRE1 & SLU1;	/*  SLU R&X CSR's*/ 
DA7.OE   = PERF &!LA1 & BUFCTL & SLU0 # PERF &!LA1 & BUFCTL & SLU1;					/*  SLU R&X CSR's*/

TBRL0    = PERF & SLU0 & LA2 & LA1 & !BUFCTL & SCTL;			/*  SLU0 */ 

DRR1     = PERF & SLU1 &!LA2 & LA1 & BUFCTL;				/*  SLU1 */ 
TBRL1    = PERF & SLU1 & LA2 & LA1 & !BUFCTL & SCTL;			/*  SLU1 */ 
 
BROM =	 !LA11 &  LA10 &  LA9 & PERF;			/* Boot Rom 173000 */

PAR0 =   LA11 &  LA10 & !LA9 & !LA8 & PERF;		/* PAR0 176160 */
MSCP =  !LA11 &  LA10 & !LA9 & PERF;			/* MSCP 172150 */
